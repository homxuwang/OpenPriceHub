import { handlePlatforms } from "./routes/platforms";
import { handleModels } from "./routes/models";
import { handlePrices } from "./routes/prices";
import { handleSubmit } from "./routes/submit";
import { handleAdmin } from "./routes/admin";
import { handleCatalog, loadCatalog } from "./routes/catalog";
import { pageMeta, renderDocument, sitemapXML } from "../shared/catalog/seo";
import type { Env } from "../shared/types";
export default {
  async fetch(request: Request, env: Env): Promise<Response> {
    const url = new URL(request.url);
    if (url.pathname.startsWith("/api/")) return handleAPI(request, env);
    const origin = env.SITE_URL || url.origin;
    if (url.pathname === "/robots.txt")
      return new Response(
        `User-agent: *\nAllow: /\nDisallow: /api/\nDisallow: /admin\nDisallow: /zh/admin\nSitemap: ${origin}/sitemap.xml\n`,
        { headers: { "Content-Type": "text/plain;charset=UTF-8" } },
      );
    if (url.pathname === "/sitemap.xml") {
      const { catalog } = await loadCatalog(env);
      return new Response(sitemapXML(catalog, origin), {
        headers: { "Content-Type": "application/xml;charset=UTF-8" },
      });
    }
    if (/\.[a-z0-9]+$/i.test(url.pathname)) return env.ASSETS.fetch(request);
    if (
      url.pathname === "/category/ai-tts" ||
      url.pathname === "/zh/category/ai-tts"
    )
      return Response.redirect(
        url.origin + url.pathname.replace("ai-tts", "ai-voice") + url.search,
        301,
      );
    const { catalog } = await loadCatalog(env);
    const meta = pageMeta(catalog, url.pathname, url.search, origin);
    const asset = await env.ASSETS.fetch(
      new Request(new URL("/", url.origin), request),
    );
    const html = renderDocument(await asset.text(), catalog, meta);
    return new Response(html, {
      status: meta.status,
      headers: {
        "Content-Type": "text/html;charset=UTF-8",
        "Cache-Control": "no-cache",
        ...(meta.noindex ? { "X-Robots-Tag": "noindex, follow" } : {}),
      },
    });
  },
};
async function handleAPI(request: Request, env: Env) {
  const url = new URL(request.url);
  const headers = {
    "Access-Control-Allow-Origin": "*",
    "Access-Control-Allow-Methods": "GET, POST, PUT, DELETE, OPTIONS",
    "Access-Control-Allow-Headers": "Content-Type, X-Admin-Key",
  };
  if (request.method === "OPTIONS") return new Response(null, { headers });
  try {
    if (url.pathname.startsWith("/api/catalog"))
      return await handleCatalog(request, env, headers);
    if (url.pathname.startsWith("/api/platforms"))
      return await handlePlatforms(request, env, headers);
    if (url.pathname.startsWith("/api/models"))
      return await handleModels(request, env, headers);
    if (url.pathname.startsWith("/api/prices"))
      return await handlePrices(request, env, headers);
    if (url.pathname === "/api/submit")
      return await handleSubmit(request, env, headers);
    if (url.pathname.startsWith("/api/admin"))
      return await handleAdmin(request, env, headers);
    return Response.json(
      { success: false, error: "Not found" },
      { status: 404, headers },
    );
  } catch (error) {
    console.error("API error", error);
    return Response.json(
      {
        success: false,
        error: "The service is temporarily unavailable. Please try again.",
      },
      { status: 500, headers },
    );
  }
}
